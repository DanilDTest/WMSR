Процедура ДобавитьПКО_Справочник_НоменклатураСегмента(ПравилаКонвертации)
	
	ПравилоКонвертации = ОбменДаннымиXDTOСервер.ИнициализироватьПравилоКонвертацииОбъекта(ПравилаКонвертации);
	ПравилоКонвертации.ИмяПКО                       = "Справочник_НоменклатураСегмента";
	ПравилоКонвертации.ОбъектДанных                 = Метаданные.РегистрыСведений.НоменклатураСегмента;
	ПравилоКонвертации.ОбъектФормата                = "Справочник.НоменклатураСегмента";
	ПравилоКонвертации.ПриКонвертацииДанныхXDTO     = "ПКО_Справочник_НоменклатураСегмента_ПриКонвертацииДанныхXDTO";
	ПравилоКонвертации.ПередЗаписьюПолученныхДанных = "ПКО_Справочник_НоменклатураСегмента_ПередЗаписьюПолученныхДанных";
	ПравилоКонвертации.ВариантИдентификации         = "ПоПолямПоиска";
	СвойстваШапки = ПравилоКонвертации.Свойства;
	ДобавитьПКС(СвойстваШапки, "", "Номенклатура", 1, "Справочник_Номенклатура_Получение");
	ДобавитьПКС(СвойстваШапки, "", "Сегмент", 1, "Справочник_СегментыНоменклатуры");
	ДобавитьПКС(СвойстваШапки, "", "Характеристика", 1, "Справочник_ХарактеристикиНоменклатуры");

	ПравилоКонвертации.ПоляПоиска.Добавить("Номенклатура,Сегмент,Характеристика");
