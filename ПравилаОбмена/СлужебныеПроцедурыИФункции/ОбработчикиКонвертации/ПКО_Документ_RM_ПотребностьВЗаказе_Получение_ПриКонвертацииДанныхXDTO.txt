Процедура ПКО_Документ_RM_ПотребностьВЗаказе_Получение_ПриКонвертацииДанныхXDTO(ДанныеXDTO, ПолученныеДанные, КомпонентыОбмена)
	ОписаниеСтрока = ОбщегоНазначения.ОписаниеТипаСтрока(64);
	ОписаниеЧисло = ОбщегоНазначения.ОписаниеТипаЧисло(15,2);
	
	ТЗПотребность = Новый ТаблицаЗначений;
	ТЗПотребность.Колонки.Добавить("НоменклатураАртикул", ОписаниеСтрока);
	ТЗПотребность.Колонки.Добавить("НоменклатураСсылка",  ОписаниеСтрока);
	ТЗПотребность.Колонки.Добавить("КоличествоПоМатрице", ОписаниеЧисло);
	ТЗПотребность.Колонки.Добавить("СвободныйОстаток",    ОписаниеЧисло);
	ТЗПотребность.Колонки.Добавить("ТоварыВПути",         ОписаниеЧисло);
	ТЗПотребность.Колонки.Добавить("КоличествоКЗаказу",   ОписаниеЧисло);  
	
	
	ПравилаЗаполненияТовары = Новый Соответствие;
	ПравилаЗаполненияТовары.Вставить("Номенклатура",        "Номенклатура");
	ПравилаЗаполненияТовары.Вставить("КоличествоПоМатрице", "КоличествоПоМатрице");
	ПравилаЗаполненияТовары.Вставить("СвободныйОстаток",    "СвободныйОстаток");
	ПравилаЗаполненияТовары.Вставить("ТоварыВПути",         "ТоварыВПути");
	ПравилаЗаполненияТовары.Вставить("КоличествоКЗаказу",   "КоличествоКЗаказу");
			
	
	Для каждого СтрокаСписка из ДанныеXDTO.Потребность Цикл
		СтруктураДанныхСтроки = ДанныеКоллекцииВВидеСтруктуры(СтрокаСписка, ПравилаЗаполненияТовары);
	 	
	 	СтрокаТаблицы = ТЗПотребность.Добавить();
	 	ЗаполнитьЗначенияСвойств(СтрокаТаблицы, СтрокаСписка);   
	 	
	 	СтрокаТаблицы.НоменклатураАртикул = ОбщегоНазначенияКлиентСервер.СвойствоСтруктуры(СтрокаСписка.Номенклатура, "Артикул", Неопределено); 
	 	
	 	СтруктураСсылки = ОбщегоНазначенияКлиентСервер.СвойствоСтруктуры(СтрокаСписка.Номенклатура, "Ссылка", Неопределено);
	 	Если НЕ СтруктураСсылки = Неопределено Тогда
	 		СтрокаТаблицы.НоменклатураСсылка = СтруктураСсылки.Значение;
	 	КонецЕсли;
	
	КонецЦикла;    
	
	ПолученныеДанные.ДополнительныеСвойства.Вставить("ТЗПотребность", ТЗПотребность);
КонецПроцедуры
