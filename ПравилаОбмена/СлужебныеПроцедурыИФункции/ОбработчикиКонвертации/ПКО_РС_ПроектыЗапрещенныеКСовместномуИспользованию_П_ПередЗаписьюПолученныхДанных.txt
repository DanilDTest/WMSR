Процедура ПКО_РС_ПроектыЗапрещенныеКСовместномуИспользованию_П_ПередЗаписьюПолученныхДанных(ПолученныеДанные, ДанныеИБ, КонвертацияСвойств, КомпонентыОбмена)
	Перечисления_Проекты = КомпонентыОбмена.ПравилаКонвертацииПредопределенныхДанных.Найти("Перечисление_RM_НаправленияДеятельности", "ИмяПКПД");
	
	
	СтруктураИзмеренийРегистра = Новый Структура;
	СтруктураИзмеренийРегистра.Вставить("ТипДокумента");
	СтруктураИзмеренийРегистра.Вставить("Проект");
	
	СоответствиеЗначений = Новый Соответствие;
	Для каждого ИзмерениеРегистра из СтруктураИзмеренийРегистра Цикл
	
	    КлючИзмерения = ИзмерениеРегистра.Ключ;
	    ЗначениеВходящее = ПолученныеДанные.ДополнительныеСвойства[ИзмерениеРегистра.Ключ];	
		
		
		Если ЗначениеВходящее = Неопределено Тогда
			Продолжить;
		ИначеЕсли КлючИзмерения = "Проект" Тогда  
		    ЗначениеВходящее = ЗначениеВходящее.Значение;
			Значение = Перечисления_Проекты.КонвертацииЗначенийПриПолучении.Получить(ЗначениеВходящее);
		ИначеЕсли КлючИзмерения = "ТипДокумента" Тогда
			Значение = ОбщегоНазначения.ИдентификаторОбъектаМетаданных(ЗначениеВходящее);
		КонецЕсли;	
		
		 ПолученныеДанные.Отбор[ИзмерениеРегистра.Ключ].Значение = Значение;
		 ПолученныеДанные.Отбор[ИзмерениеРегистра.Ключ].Использование = Истина;  
		 
		 СоответствиеЗначений.Вставить(ЗначениеВходящее, Значение);
	КонецЦикла;
	       
	СтруктураСсылочныхПолей = СтруктураИзмеренийРегистра;
	    
		
	// Добавление записей в набор записей.		
	Если ПолученныеДанные.ДополнительныеСвойства.Записи <> Неопределено Тогда
		
		Для Каждого Запись Из ПолученныеДанные.ДополнительныеСвойства.Записи Цикл
			
			СтрокаЗаписи = ПолученныеДанные.Добавить();   
	
					
			Для каждого ИзмерениеРегистра из СтруктураСсылочныхПолей Цикл
				
				
			    КлючИзмерения = ИзмерениеРегистра.Ключ;
			    ЗначениеВходящее = ПолученныеДанные.ДополнительныеСвойства[ИзмерениеРегистра.Ключ];	
				
				Значение = СоответствиеЗначений[ЗначениеВходящее];
				
				Если ЗначениеВходящее = Неопределено Тогда
					Продолжить;                           
				ИначеЕсли НЕ Значение = Неопределено Тогда
					//Уже определили 
				ИначеЕсли КлючИзмерения = "Проект" Тогда  
				
					 ЗначениеВходящее = ЗначениеВходящее.Значение;
					Значение = Перечисления_Проекты.КонвертацииЗначенийПриПолучении.Получить(ЗначениеВходящее);
				ИначеЕсли КлючИзмерения = "ТипДокумента" Тогда
					Значение = ОбщегоНазначения.ИдентификаторОбъектаМетаданных(ЗначениеВходящее);
				КонецЕсли;	
				
				СоответствиеЗначений.Вставить(ЗначениеВходящее, Значение);			
					
				СтрокаЗаписи[ИзмерениеРегистра.Ключ] = Значение;
		 		
			КонецЦикла;			
				
		КонецЦикла;
		
	КонецЕсли;
КонецПроцедуры
