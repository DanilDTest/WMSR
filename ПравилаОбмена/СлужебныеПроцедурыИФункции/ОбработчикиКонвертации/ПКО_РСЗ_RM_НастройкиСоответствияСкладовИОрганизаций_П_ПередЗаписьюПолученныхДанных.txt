Процедура ПКО_РСЗ_RM_НастройкиСоответствияСкладовИОрганизаций_П_ПередЗаписьюПолученныхДанных(ПолученныеДанные, ДанныеИБ, КонвертацияСвойств, КомпонентыОбмена)
	Справочник_Склады_Получение     = RM_ОбменДаннымиXDTOСервер.ПКОПоИмени(КомпонентыОбмена,"Справочник_Склады_Получение");
	Справочник_Организации_Получение = RM_ОбменДаннымиXDTOСервер.ПКОПоИмени(КомпонентыОбмена,"Справочник_Организации_Получение");
	
	ПолученныеДанные.Отбор.Период.Значение      = ПолученныеДанные.ДополнительныеСвойства.Период;
	ПолученныеДанные.Отбор.Период.Использование = Истина;    
	                                                           
	Склад = RM_ОбменДаннымиXDTOСервер.СтруктураОбъектаXDTOВДанныеИБ(КомпонентыОбмена, 
			ПолученныеДанные.ДополнительныеСвойства.Склад, 
			Справочник_Склады_Получение, 
			"ПолучитьСсылку");
	                                                           
	ПолученныеДанные.Отбор.Склад.Значение      = Склад;
	ПолученныеДанные.Отбор.Склад.Использование = Истина;
	
	
	// Добавление записей в набор записей.		
	Если ПолученныеДанные.ДополнительныеСвойства.Записи <> Неопределено Тогда
		
		Для Каждого Строка Из ПолученныеДанные.ДополнительныеСвойства.Записи Цикл
			
			СтрокаЗаписи          = ПолученныеДанные.Добавить();
			СтрокаЗаписи.Период = ПолученныеДанные.ДополнительныеСвойства.Период;
			СтрокаЗаписи.Склад  = Склад;                   
			
			Организация = RM_ОбменДаннымиXDTOСервер.СтруктураОбъектаXDTOВДанныеИБ(КомпонентыОбмена, 
				Строка.Организация, 
				Справочник_Организации_Получение, 
				"ПолучитьСсылку");
			                               						                               
			СтрокаЗаписи.Организация = Организация;         
						
		КонецЦикла;
		
	КонецЕсли;
КонецПроцедуры
