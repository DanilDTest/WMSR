Процедура ПКО_Справочник_RM_ИзображенияНоменклатуры_Получение_ПередЗаписьюПолученныхДанных(ПолученныеДанные, ДанныеИБ, КонвертацияСвойств, КомпонентыОбмена)
	Справочник_Номенклатура_Получение     = RM_ОбменДаннымиXDTOСервер.ПКОПоИмени(КомпонентыОбмена,"Справочник_Номенклатура_Получение");
	Перечисление_РасширенияФайлов    	  = КомпонентыОбмена.ПравилаКонвертацииПредопределенныхДанных.Найти("Перечисление_РасширенияФайлов");
	Перечисление_ТипыИзображенийНоменклатуры    = КомпонентыОбмена.ПравилаКонвертацииПредопределенныхДанных.Найти("Перечисление_RM_ТипыИзображенийНоменклатуры");
	      
	Номенклатура = RM_ОбменДаннымиXDTOСервер.СтруктураОбъектаXDTOВДанныеИБ(КомпонентыОбмена, 
				ПолученныеДанные.ДополнительныеСвойства.Номенклатура, 
				Справочник_Номенклатура_Получение, 
				"ПолучитьСсылку");
			                                             
	ПолученныеДанные.Отбор.Номенклатура.Значение      = Номенклатура;
	ПолученныеДанные.Отбор.Номенклатура.Использование = Истина;    
	                         
	УИДИзображения = Неопределено;
	ПолученныеДанные.ДополнительныеСвойства.Свойство("GUIDНоменклатураПрисоединенныеФайлы", УИДИзображения);                       
	
	Если НЕ УИДИзображения = Неопределено Тогда     
		ПолученныеДанные.Отбор.GUIDНоменклатураПрисоединенныеФайлы.Значение      = УИДИзображения;
		ПолученныеДанные.Отбор.GUIDНоменклатураПрисоединенныеФайлы.Использование = Истина;      
	КонецЕсли;   
	
	// Добавление записей в набор записей.		
	Если ПолученныеДанные.ДополнительныеСвойства.Записи <> Неопределено Тогда
		
		Для Каждого Строка Из ПолученныеДанные.ДополнительныеСвойства.Записи Цикл
			
			СтрокаЗаписи          = ПолученныеДанные.Добавить();
	   						                               
			СтрокаЗаписи.Номенклатура = Номенклатура; 
			
			Если УИДИзображения = Неопределено 
				И ОбщегоНазначенияКлиентСервер.ЕстьРеквизитИлиСвойствоОбъекта(Строка, "GUIDНоменклатураПрисоединенныеФайлы") Тогда
				стрУИДИзображения = Новый УникальныйИдентификатор(Строка.GUIDНоменклатураПрисоединенныеФайлы);
				СтрокаЗаписи.GUIDНоменклатураПрисоединенныеФайлы = стрУИДИзображения;   
			Иначе 
			    СтрокаЗаписи.GUIDНоменклатураПрисоединенныеФайлы = УИДИзображения; 
			КонецЕсли;
	  	       	
			Если ОбщегоНазначенияКлиентСервер.ЕстьРеквизитИлиСвойствоОбъекта(Строка, "URLАдресИзображения") Тогда
				СтрокаЗаписи.URLАдресИзображения = Строка.URLАдресИзображения;
			КонецЕсли;   
									
			Если ОбщегоНазначенияКлиентСервер.ЕстьРеквизитИлиСвойствоОбъекта(Строка, "ТипИзображения") Тогда
				ТипИзображения = ОбщегоНазначенияКлиентСервер.СвойствоСтруктуры(Строка.ТипИзображения, "Значение"); 
				
				ТипИзображенияЗначение = Перечисление_ТипыИзображенийНоменклатуры.КонвертацииЗначенийПриПолучении.Получить(ТипИзображения);	
				СтрокаЗаписи.ТипИзображения = ТипИзображенияЗначение;
			КонецЕсли;   
			
			Если ОбщегоНазначенияКлиентСервер.ЕстьРеквизитИлиСвойствоОбъекта(Строка, "РасширениеИзображения") Тогда
				РасширениеИзображения = ОбщегоНазначенияКлиентСервер.СвойствоСтруктуры(Строка.РасширениеИзображения, "Значение"); 
				
				РасширениеИзображенияЗначение = Перечисление_РасширенияФайлов.КонвертацииЗначенийПриПолучении.Получить(РасширениеИзображения);	
				СтрокаЗаписи.РасширениеИзображения = РасширениеИзображенияЗначение;
			КонецЕсли; 
	
	                    
	                    
		КонецЦикла;
		
	КонецЕсли;
КонецПроцедуры
