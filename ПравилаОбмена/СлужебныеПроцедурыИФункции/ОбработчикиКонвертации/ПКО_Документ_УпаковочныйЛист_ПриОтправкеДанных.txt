Процедура ПКО_Документ_УпаковочныйЛист_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	Товары = ДанныеИБ.Товары.Выгрузить();
	ДанныеXDTO.Вставить("Товары", Товары);
	
	ПопутныйГруз = ДанныеИБ.RM_ПопутныйГруз.Выгрузить();
	ДанныеXDTO.Вставить("ПопутныйГруз", ПопутныйГруз);      
	
	ШтрихкодыУпаковок = ДанныеИБ.RM_ШтрихкодыУпаковок.Выгрузить();    
	ШтрихкодыУпаковок.Колонки.Добавить("RM_КлючСвязи");
	
	Для каждого Строка из ШтрихкодыУпаковок Цикл
		Строка.RM_КлючСвязи = Строка.КлючСвязи;
	КонецЦикла;
	ДанныеXDTO.Вставить("АкцизныеМарки", ШтрихкодыУпаковок);        
	                  
	ЗаказНаПеремещение = Неопределено;                  
	Если ЗначениеЗаполнено(ДанныеИБ.RM_ЗаказНаПеремещение) Тогда
		ЗаказНаПеремещение = ДанныеИБ.RM_ЗаказНаПеремещение;
	ИначеЕсли ТипЗнч(ДанныеИБ.RM_ДокументОснование) = Тип("ДокументСсылка.ПеремещениеТоваров") Тогда  
		ЗаказНаПеремещение = ОбщегоНазначения.ЗначениеРеквизитаОбъекта(ДанныеИБ.RM_ДокументОснование, "ЗаказНаПеремещение");
	КонецЕсли;                         
	
	
	Если ЗначениеЗаполнено(ЗаказНаПеремещение) Тогда
		ДанныеXDTO.Вставить("ЗаказНаПеремещение", ЗаказНаПеремещение); 
	КонецЕсли;  
	
	RM_ДокументОснование = ДанныеИБ.RM_ДокументОснование;
	Если RM_ДокументОснование <> Неопределено Тогда
	    
	    Если ТипЗнч(RM_ДокументОснование) = Тип("ДокументСсылка.ПеремещениеТоваров") Тогда
	       ИмяПКО = "Документ_ПеремещениеТоваров_Отправка";
	    КонецЕсли;   
	    
	    Если ЗначениеЗаполнено(ИмяПКО) Тогда
	    	ДанныеXDTO.Вставить("ДокументОснование", Новый Структура("Значение, ИмяПКО", RM_ДокументОснование, ИмяПКО));  
	    КонецЕсли;
	КонецЕсли;                    
	
	СтруктураДополнительныхРеквизитов = Новый Структура;
	СтруктураДополнительныхРеквизитов.Вставить("RM_Смешанный");
	
	ЗаполнитьЗначенияСвойств(СтруктураДополнительныхРеквизитов, ДанныеИБ);  
	ВыгрузитьДополнительныеРеквизитыСвойствоСтрокой(СтруктураДополнительныхРеквизитов, ДанныеXDTO);
КонецПроцедуры
