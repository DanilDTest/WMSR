Процедура ПКО_Документ_ПередачаТоваровМеждуОрганизациями_Получен_ПриКонвертацииДанныхXDTO(ДанныеXDTO, ПолученныеДанные, КомпонентыОбмена)
	ПолученныеДанные.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.РеализацияТоваровВДругуюОрганизацию; 
	ПолученныеДанные.ПорядокОплаты = Перечисления.ПорядокОплатыПоСоглашениям.РасчетыВРубляхОплатаВРублях;
	ПолученныеДанные.ФормаОплаты = Перечисления.ФормыОплаты.Наличная;
	           
	ДокументОснование = Неопределено;           
	Если ДанныеXDTO.Свойство("ДокументОснование", ДокументОснование) Тогда
	    ИмяПКО = "Документ_ПеремещениеТоваров_Получение";
	    Если Найти(ДокументОснование.ТипЗначения, "КорректировкаПоАктуРасхожденийПослеПеремещения") > 0 Тогда
	        ИмяПКО = "Документ_КорректировкаПоАктуРасхожденийПП_Получени";
	    КонецЕсли;
	    ПолученныеДанные.ДополнительныеСвойства.Вставить("RM_ДокументОснование", Новый Структура("Значение, ИмяПКО", ДокументОснование, ИмяПКО));
	КонецЕсли;      
	
	ПравилаЗаполнения = Новый Соответствие;
	ПравилаЗаполнения.Вставить("ШтрихкодУпаковки", "ШтрихкодУпаковки");   
	 
	МассивСтрокКМ = Новый Массив;
	
	Если ДанныеXDTO.Свойство("АкцизныеМарки") Тогда  
		Для Каждого Строка Из ДанныеXDTO.АкцизныеМарки Цикл  
		                            
			СтруктураДанныхСтроки = ДанныеКоллекцииВВидеСтруктуры(Строка, ПравилаЗаполнения); 
			МассивСтрокКМ.Добавить(СтруктураДанныхСтроки);
		КонецЦикла;	       	
	КонецЕсли;   
	
	ПолученныеДанные.ДополнительныеСвойства.Вставить("ШтрихкодыУпаковок", МассивСтрокКМ);
	                   
	
	     
	ПравилаЗаполнения = Новый Соответствие;
	ПравилаЗаполнения.Вставить("Номенклатура", "Номенклатура");   
	ПравилаЗаполнения.Вставить("Упаковка", "Упаковка");   
	ПравилаЗаполнения.Вставить("СуммаНДС", "СуммаНДС");   
	ПравилаЗаполнения.Вставить("СтавкаНДС", "СтавкаНДС");
	ПравилаЗаполнения.Вставить("КоличествоУпаковок", "КоличествоУпаковок");   
	ПравилаЗаполнения.Вставить("Количество", "Количество");
	ПравилаЗаполнения.Вставить("Сумма", "Сумма");   
	ПравилаЗаполнения.Вставить("Цена", "Цена");      
	ПравилаЗаполнения.Вставить("ИдентификаторСтроки", "ИдентификаторСтроки");   
	
	МассивСтрокТовары = Новый Массив;
	
	Если ДанныеXDTO.Свойство("Товары") Тогда  
		Для Каждого Строка Из ДанныеXDTO.Товары Цикл  
		                            
			СтруктураДанныхСтроки = ДанныеКоллекцииВВидеСтруктуры(Строка, ПравилаЗаполнения); 
			МассивСтрокТовары.Добавить(СтруктураДанныхСтроки);
		КонецЦикла;	       	
	КонецЕсли;   
	
	ПолученныеДанные.ДополнительныеСвойства.Вставить("Товары", МассивСтрокТовары);
	
	       
	
	ПравилаЗаполнения = Новый Соответствие;
	ПравилаЗаполнения.Вставить("СтранаПроисхождения", "СтранаПроисхождения");   
	ПравилаЗаполнения.Вставить("НомерГТД", "НомерГТД");
	
	Если ДанныеXDTO.Свойство("ВидыЗапасов") Тогда  
		
		ИндексСтроки = 0; 
		ШаблонГТД = "[НомерГТД]; [Наименование]; [Код]"; //"[НомерГТД]; [НаименованиеСтраны]; [КодСтраны]";
		
		Для Каждого Строка Из ДанныеXDTO.ВидыЗапасов Цикл                              
			НомерГТД = "";
			СтруктураДанныхСтроки = ДанныеКоллекцииВВидеСтруктуры(Строка, ПравилаЗаполнения);
			
			Если СтруктураДанныхСтроки.Свойство("НомерГТД") Тогда  
				НомерГТД = СокрП(СтруктураДанныхСтроки.НомерГТД);
			КонецЕсли;
			
			Если ПустаяСтрока(НомерГТД) Тогда
				Продолжить;
			КонецЕсли;
			
			СтрокаТЧ = ПолученныеДанные.ВидыЗапасов[ИндексСтроки];  
			
			ДанныеГТД = Новый Структура("Код, Наименование, НомерГТД");  
			ДанныеГТД.НомерГТД = НомерГТД;
			
			Если СтруктураДанныхСтроки.Свойство("СтранаПроисхождения") 
				И СтруктураДанныхСтроки.СтранаПроисхождения.Свойство("ДанныеКлассификатора") Тогда 
				
				ЗаполнитьЗначенияСвойств(ДанныеГТД, СтруктураДанныхСтроки.СтранаПроисхождения.ДанныеКлассификатора);        		
			КонецЕсли;               
			
			
			НомерГТДПредставление = СтроковыеФункцииКлиентСервер.ВставитьПараметрыВСтроку(ШаблонГТД, ДанныеГТД);
			СтрокаТЧ.НомерГТД = НомерГТДПредставление;     
					
			ИндексСтроки = ИндексСтроки + 1;
		КонецЦикла;	
	КонецЕсли;
КонецПроцедуры
