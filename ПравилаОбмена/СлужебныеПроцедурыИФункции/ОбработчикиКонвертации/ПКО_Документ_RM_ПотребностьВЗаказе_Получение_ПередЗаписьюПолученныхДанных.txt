Процедура ПКО_Документ_RM_ПотребностьВЗаказе_Получение_ПередЗаписьюПолученныхДанных(ПолученныеДанные, ДанныеИБ, КонвертацияСвойств, КомпонентыОбмена)
	ТЗПотребность = Неопределено;
	ПолученныеДанные.Потребность.Очистить();
	
	
	Если ПолученныеДанные.ДополнительныеСвойства.Свойство("ТЗПотребность", ТЗПотребность) Тогда   
	
	 
	    ДанныеДляЗаписи = ПолученныеДанные;
	    
		Запрос = Новый Запрос;
		Запрос.Текст = 
			"ВЫБРАТЬ
			|	ТЗВходящая.НоменклатураСсылка КАК НоменклатураСсылкаИдентификатор,
			|	ТЗВходящая.НоменклатураАртикул КАК НоменклатураАртикул,
			|	ТЗВходящая.КоличествоПоМатрице КАК КоличествоПоМатрице,
			|	ТЗВходящая.СвободныйОстаток КАК СвободныйОстаток,
			|	ТЗВходящая.ТоварыВПути КАК ТоварыВПути,
			|	ТЗВходящая.КоличествоКЗаказу КАК КоличествоКЗаказу,
			|	АВТОНОМЕРЗАПИСИ() КАК НомерЗаписи
			|ПОМЕСТИТЬ Входящая
			|ИЗ
			|	&ТЗВходящая КАК ТЗВходящая
			|;
			|
			|////////////////////////////////////////////////////////////////////////////////
			|ВЫБРАТЬ
			|	МАКСИМУМ(Номенклатура.Ссылка) КАК Ссылка,
			|	Входящая.НоменклатураСсылкаИдентификатор КАК НоменклатураСсылкаИдентификатор
			|ПОМЕСТИТЬ СопоставлениеСсылок
			|ИЗ
			|	Справочник.Номенклатура КАК Номенклатура
			|		ВНУТРЕННЕЕ СОЕДИНЕНИЕ Входящая КАК Входящая
			|		ПО Номенклатура.Артикул = Входящая.НоменклатураАртикул
			|			И (НЕ Входящая.НоменклатураАртикул = """")
			|
			|СГРУППИРОВАТЬ ПО
			|	Входящая.НоменклатураСсылкаИдентификатор
			|;
			|
			|////////////////////////////////////////////////////////////////////////////////
			|ВЫБРАТЬ
			|	Входящая.КоличествоПоМатрице КАК КоличествоПоМатрице,
			|	Входящая.СвободныйОстаток КАК СвободныйОстаток,
			|	Входящая.ТоварыВПути КАК ТоварыВПути,
			|	Входящая.КоличествоКЗаказу КАК КоличествоКЗаказу,
			|	Входящая.НоменклатураСсылкаИдентификатор КАК НоменклатураСсылкаИдентификатор,
			|	Входящая.НоменклатураАртикул КАК НоменклатураАртикул,
			|	ЕСТЬNULL(СопоставлениеСсылок.Ссылка, НЕОПРЕДЕЛЕНО) КАК Ссылка,
			|	Входящая.НомерЗаписи КАК НомерЗаписи
			|ИЗ
			|	Входящая КАК Входящая
			|		ЛЕВОЕ СОЕДИНЕНИЕ СопоставлениеСсылок КАК СопоставлениеСсылок
			|		ПО Входящая.НоменклатураСсылкаИдентификатор = СопоставлениеСсылок.НоменклатураСсылкаИдентификатор
			|
			|УПОРЯДОЧИТЬ ПО
			|	НомерЗаписи";
	
		Запрос.УстановитьПараметр("ТЗВходящая", ТЗПотребность);
	
	
		РезультатЗапроса = Запрос.Выполнить();	
		Выборка = РезультатЗапроса.Выбрать();	
			Пока Выборка.Следующий() Цикл       
	
			СсылкаНоменклатуры = Выборка.Ссылка;
			Если СсылкаНоменклатуры = Неопределено 
				И НЕ ПустаяСтрока(Выборка.НоменклатураСсылкаИдентификатор) Тогда 
				УИДСсылки = Новый УникальныйИдентификатор(Выборка.НоменклатураСсылкаИдентификатор);
				СсылкаНоменклатуры = Справочники.Номенклатура.ПолучитьСсылку(УИДСсылки);		
			КонецЕсли;                                                                  
	
			Стр = ДанныеДляЗаписи.Потребность.Добавить();
			ЗаполнитьЗначенияСвойств(Стр, Выборка);
			Стр.Номенклатура = СсылкаНоменклатуры;
			
		КонецЦикла;
	                   
	КонецЕсли;
КонецПроцедуры
