Процедура ПКО_Документ_РасходныйОрдерНаТовары_Отправка_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	Если СтекВыгрузки.Количество() > 1 Тогда
		Возврат;
	КонецЕсли;              
	  
	Запрос = Новый Запрос;
	Запрос.Текст = 
	"ВЫБРАТЬ
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Серия КАК Серия,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.СтатусУказанияСерий КАК СтатусУказанияСерий,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Распоряжение КАК РаспоряжениеСсылка,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Количество КАК Количество,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Номенклатура КАК Номенклатура,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Характеристика КАК Характеристика,
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.RM_УпаковочныйЛист КАК УпаковочныйЛист
	|ИЗ
	|	Документ.РасходныйОрдерНаТовары.ТоварыПоРаспоряжениям КАК РасходныйОрдерНаТоварыТоварыПоРаспоряжениям
	|ГДЕ
	|	РасходныйОрдерНаТоварыТоварыПоРаспоряжениям.Ссылка = &Ссылка";
	
	
	Запрос.УстановитьПараметр("Ссылка", ДанныеИБ.Ссылка);
	РезультатЗапроса = Запрос.Выполнить();
	
	ТоварыПоРаспоряжениям = РезультатЗапроса.Выгрузить(); 
	
	ДокументОснование = ТоварыПоРаспоряжениям[0].РаспоряжениеСсылка;  
	ТипЗначенияОснования = ТипЗнч(ДокументОснование);                     
	ИмяПКОРаспоряжения = ""; 
	Если ТипЗначенияОснования = Тип("ДокументСсылка.ПеремещениеТоваров") Тогда
		ИмяПКОРаспоряжения = "Документ_ПеремещениеТоваров_Отправка";  
	ИначеЕсли ТипЗначенияОснования = Тип("ДокументСсылка.ЗаказКлиента") Тогда
		ИмяПКОРаспоряжения = "Документ_ЗаказКлиента_Отправка";
	ИначеЕсли ТипЗначенияОснования = Тип("ДокументСсылка.ЗаказНаПеремещение") Тогда
		ИмяПКОРаспоряжения = "Документ_ЗаказНаПеремещение_Отправка";  
	ИначеЕсли ТипЗначенияОснования = Тип("ДокументСсылка.ЗаказНаСборку") Тогда
		ИмяПКОРаспоряжения = "Документ_ЗаказНаСборку_Отправка";
	
	КонецЕсли;
	
	Если НЕ ИмяПКОРаспоряжения = "" Тогда
		ДанныеXDTO.Вставить("ДокументОснование", Новый Структура("Значение, ИмяПКО", ДокументОснование, ИмяПКОРаспоряжения)); 
	КонецЕсли;  
	
	Если НЕ ИмяПКОРаспоряжения = "" Тогда
		ТоварыПоРаспоряжениям.Колонки.Добавить("Распоряжение");			
		Для каждого СтрокаРаспоряжения из ТоварыПоРаспоряжениям Цикл   
			
			СтрокаРаспоряжения.Распоряжение = Новый Структура("Значение, ИмяПКО", СтрокаРаспоряжения.РаспоряжениеСсылка, ИмяПКОРаспоряжения);	
		КонецЦикла;
	КонецЕсли;
	ДанныеXDTO.Вставить("ТоварыПоРаспоряжениям", ТоварыПоРаспоряжениям);//Из ПОД
	
	ТипПолучателя = ТипЗнч(ДанныеИБ.Получатель);
	ИмяПКОПолучатель = "";
	
	Если ТипПолучателя = Тип("СправочникСсылка.Склады") Тогда
		ИмяПКОПолучатель = "Справочник_Склады_Отправка";
	ИначеЕсли ТипПолучателя = Тип("СправочникСсылка.ФизическиеЛица") Тогда
		ИмяПКОПолучатель = "Справочник_ФизическиеЛица_Отправка";
	ИначеЕсли ТипПолучателя = Тип("СправочникСсылка.Партнеры") Тогда
		ИмяПКОПолучатель = "Справочник_Партнер";      
	КонецЕсли;	
	
	Если НЕ ИмяПКОПолучатель = "" Тогда
		ДанныеXDTO.Вставить("СкладПолучатель", Новый Структура("Значение, ИмяПКО", ДанныеИБ.Получатель, ИмяПКОПолучатель)); 
	КонецЕсли; 
	  
	Если ДанныеИБ.ДополнительныеСвойства.Свойство("Товары") Тогда  
		ДанныеXDTO.Вставить("Товары", ДанныеИБ.ДополнительныеСвойства.Товары);//Из ПОД
		ДанныеXDTO.Вставить("АкцизныеМарки",  ДанныеИБ.ДополнительныеСвойства.ШтрихкодыУпаковок); //Из ПОД   
	КонецЕсли;
КонецПроцедуры
