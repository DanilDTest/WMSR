Процедура ПКО_Докумен_RM_Приемка_Отправка_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	RM_ПриемкаТовары.Ссылка КАК Ссылка,
		|	RM_ПриемкаТовары.НомерСтроки КАК НомерСтроки,
		|	RM_ПриемкаТовары.Номенклатура КАК Номенклатура,
		|	RM_ПриемкаТовары.Характеристика КАК Характеристика,
		|	RM_ПриемкаТовары.Упаковка КАК Упаковка,
		|	RM_ПриемкаТовары.КоличествоПлан КАК КоличествоПлан,
		|	RM_ПриемкаТовары.КоличествоФакт КАК КоличествоФакт,
		|	RM_ПриемкаТовары.Ключ КАК КлючУИД
		|ИЗ
		|	Документ.RM_Приемка.Товары КАК RM_ПриемкаТовары
		|ГДЕ
		|	RM_ПриемкаТовары.Ссылка = &Ссылка";
	
	Запрос.УстановитьПараметр("Ссылка", ДанныеИБ.Ссылка);   
	
	ТаблицаРезультат = Запрос.Выполнить().Выгрузить();
	ТаблицаРезультат.Колонки.Добавить("Ключ");
		
	Для каждого СтрокаТаблицы из ТаблицаРезультат Цикл 
		СтрокаТаблицы.Ключ = Строка(СтрокаТаблицы.КлючУИД);	
	КонецЦикла;
	
	ДанныеXDTO.Вставить("Товары", ТаблицаРезультат);
	   
	ТипОснования = ТипЗнч(ДанныеИБ.ДокументОснование);
	ИмяПКО = "";                                           
	Если ТипОснования = Тип("ДокументСсылка.ЗаказПоставщику") Тогда
		ИмяПКО = "Документ_ЗаказПоставщику_Отправка";  
	ИначеЕсли ТипОснования = Тип("ДокументСсылка.ПеремещениеТоваров") Тогда
		 ИмяПКО = "Документ_ПеремещениеТоваров_Отправка";  
	ИначеЕсли ТипОснования = Тип("ДокументСсылка.ЗаказНаПеремещение") Тогда
		 ИмяПКО = "Документ_ЗаказНаПеремещение_Отправка"; 
	КонецЕсли;
	
	Если НЕ ИмяПКО = "" Тогда
		ДанныеXDTO.Вставить("ДокументОснование", Новый Структура("Значение, ИмяПКО", ДанныеИБ.ДокументОснование, ИмяПКО));	
	КонецЕсли;
КонецПроцедуры
