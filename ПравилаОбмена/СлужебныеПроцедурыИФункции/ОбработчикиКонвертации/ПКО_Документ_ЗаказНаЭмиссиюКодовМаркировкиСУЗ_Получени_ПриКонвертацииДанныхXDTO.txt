Процедура ПКО_Документ_ЗаказНаЭмиссиюКодовМаркировкиСУЗ_Получени_ПриКонвертацииДанныхXDTO(ДанныеXDTO, ПолученныеДанные, КомпонентыОбмена)
	Если ДанныеXDTO.Свойство("ДокументОснование") Тогда 
	
	    ИмяПКО = "";
		ДокументОснование = ДанныеXDTO.ДокументОснование; 
		
	    Если Найти(ДокументОснование.ТипЗначения, "ЗаказПоставщику") > 0 Тогда
	        ИмяПКО = "Документ_ЗаказПоставщику_Получение";
	    КонецЕсли;    
	    
	    Если НЕ ПустаяСтрока(ИмяПКО) Тогда
	    	ПолученныеДанные.ДополнительныеСвойства.Вставить("ДокументОснование", Новый Структура("Значение, ИмяПКО", ДокументОснование, ИмяПКО));   
	    КонецЕсли;
	КонецЕсли;          
	
	ЗагрузитьДополнительныеРеквизиты(ПолученныеДанные, ДанныеXDTO, КомпонентыОбмена);
