Процедура ПКО_Документ_RM_ОстаткиДляВыгрузкиВБУС_Отправка_ПриОтправкеДанных(ДанныеИБ, ДанныеXDTO, КомпонентыОбмена, СтекВыгрузки)
	Запрос = Новый Запрос;
	Запрос.Текст = 
	"ВЫБРАТЬ
	|	RM_ОстаткиДляВыгрузкиВБУСТовары.Номенклатура КАК НоменклатураСсылка,
	|	RM_ОстаткиДляВыгрузкиВБУСТовары.Номенклатура.Артикул КАК НоменклатураАртикул,
	|	RM_ОстаткиДляВыгрузкиВБУСТовары.Номенклатура.Наименование КАК НоменклатураНаименование,
	|	RM_ОстаткиДляВыгрузкиВБУСТовары.Остаток КАК Остаток
	|ИЗ
	|	Документ.RM_ОстаткиДляВыгрузкиВБУС.Товары КАК RM_ОстаткиДляВыгрузкиВБУСТовары
	|ГДЕ
	|	RM_ОстаткиДляВыгрузкиВБУСТовары.Ссылка = &Ссылка";
	
	Запрос.УстановитьПараметр("Ссылка", ДанныеИБ.Ссылка);
	
	Товары = Запрос.Выполнить().Выгрузить();
	Товары.Колонки.Добавить("Номенклатура");
	
	Для Каждого СтрокаТовары Из Товары Цикл
		
		ДанныеНоменклатуры = Новый Структура;
		ДанныеНоменклатуры.Вставить("Ссылка",		СтрокаТовары.НоменклатураСсылка);
		ДанныеНоменклатуры.Вставить("Артикул",		СтрокаТовары.НоменклатураАртикул);
		ДанныеНоменклатуры.Вставить("Наименование",	СтрокаТовары.НоменклатураНаименование);
		
		СтрокаТовары.Номенклатура = Новый Структура("Значение, ИмяПКО", ДанныеНоменклатуры, "Справочник_Номенклатура_Из_Структуры")
	
	КонецЦикла;
	
	Товары.Колонки.Удалить("НоменклатураСсылка");
	Товары.Колонки.Удалить("НоменклатураАртикул");
	Товары.Колонки.Удалить("НоменклатураНаименование");
	
	ДанныеXDTO.Вставить("Товары", Товары);
	
	ДанныеИБ.ОтразитьОстаткиТоваровБУС();
КонецПроцедуры
